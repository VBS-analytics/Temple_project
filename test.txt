  phone_number: string;
  password: string;
};

const highlights = [
  'Daily darshan updates and priority booking',
  'Manage seva schedules and donations in one place',
  'Personalized spiritual content curated for you',
] as const;

const LoginPage = () => {

  const setAuth = useAuthStore((state) => state.setAuth);
  const {
    register,
    handleSubmit,
    formState: { errors, isSubmitting },
  } = useForm<FormValues>({
    defaultValues: { phone_number: '', password: '' },

  const onSubmit = async (values: FormValues) => {
    setApiError(null);

      navigate('/dashboard');

      const detail = error?.response?.data?.detail ?? 'Unable to login. Please check your credentials.';
      setApiError(detail);
    }
